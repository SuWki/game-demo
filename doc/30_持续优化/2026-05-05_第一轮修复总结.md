#### 4. 结算页信息不足

- **修复**: 在RunResult中添加selectedUpgrades字段，结算页展示强化历史
- **结果**: 构建通过 ✅

#### 5. 暴击视觉特效不明显
- **问题**: 暴击橙色爆闪不明显，无法区分是否暴击

- **结果**: 构建通过 ✅

#### 6. Boss和精英护卫AI不智能
- **问题**: 护卫直接追击玩家，不会保护Boss，导致Boss成为靶子
- **修复**: 实现专门的护卫AI系统（阵型保持、拦截玩家、智能攻击）
- **结果**: 构建通过 ✅

private applyModifiers(modifiers: Partial<PlayerStats>): void {
  const maxHpGain = modifiers.maxHp ?? 0;
  this.state.stats.maxHp += maxHpGain;
  // 加生命上限时同时回复相同数值的血量
  if (maxHpGain > 0) {
    this.state.stats.hp = clamp(this.state.stats.hp + maxHpGain, 0, this.state.stats.maxHp);
  } else {
    this.state.stats.hp = clamp(this.state.stats.hp, 0, this.state.stats.maxHp);

  // 找到Boss/精英本体
  const boss = this.state.enemies.find(e => 
    e.role === 'boss' || e.role === 'elite'
  );
  if (!boss) return;

  // 计算阵型位置
  const formationAngle = (enemy.id * 1.618) % (Math.PI * 2);
  const formationDist = 150;
  const formationX = boss.x + Math.cos(formationAngle) * formationDist;
  const formationY = boss.y + Math.sin(formationAngle) * formationDist;

  // 计算拦截位置
  const playerDist = Math.hypot(
    this.state.player.x - boss.x,
    this.state.player.y - boss.y
  );
  const interceptWeight = playerDist < 200 ? 0.6 : 0.2;

  // 根据护卫类型混合移动权重
  let formationWeight = 0.5;
  if (enemy.archetype === 'ranged') formationWeight = 0.7;
  if (enemy.archetype === 'skirmisher') formationWeight = 0.4;

                  interceptX * interceptWeight +
